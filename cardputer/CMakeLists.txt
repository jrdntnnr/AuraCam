cmake_minimum_required(VERSION 3.25)
project(auracam_cardputer VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
include(FetchContent)
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
FetchContent_Declare(lvgl GIT_REPOSITORY https://github.com/lvgl/lvgl.git GIT_TAG v9.5.0 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(lvgl)
add_executable(aura-cardputer main.cpp)
target_link_libraries(aura-cardputer PRIVATE lvgl m pthread)
install(TARGETS aura-cardputer DESTINATION bin)
